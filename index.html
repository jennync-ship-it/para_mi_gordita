Al crear una carpeta, actualiza la pantalla al instante window.createFolder = async function() { const name = $("newFolderName").value.trim(); if (!name) return; const { error } = await supabaseClient.from('folders').insert([{ name }]); if (error) { showMessage("Error al crear la carpeta: " + error.message, "error"); return; } $("newFolderName").value = ""; $("adminFolderContainer").style.display = 'none'; showMessage("¡Carpeta creada con éxito!", "success"); // ⚡ Actualización inmediata en tiempo real await loadFoldersAndResources(); if (currentUserRole === 'admin') await loadAdminManagementLists(); }; // 2. Al subir un archivo, refresca la vista inmediatamente sin demoras window.uploadResource = async function() { const title = $("resTitle").value.trim(); const description = $("resDesc").value.trim(); const folder_id = $("resFolderSelect").value; const fileInput = $("resFile"); if (!folder_id) { showMessage("Por favor, selecciona una carpeta de destino.", "error"); return; } if (!fileInput.files || fileInput.files.length === 0) { showMessage("Por favor, selecciona un archivo para subir.", "error"); return; } const file = fileInput.files[0]; const fileExt = file.name.split('.').pop(); const fileName = `${Date.now()}-${Math.random().toString(36).substring(2, 9)}.${fileExt}`; const filePath = `${fileName}`; showMessage("Subiendo archivo a la nube... Por favor espera.", "info"); const { error: uploadError } = await supabaseClient.storage .from('documents') .upload(filePath, file); if (uploadError) { showMessage("Error al subir el archivo al Storage: " + uploadError.message, "error"); return; } const { data: publicURLData } = supabaseClient.storage .from('documents') .getPublicUrl(filePath); const file_url = publicURLData.publicUrl; const { error: dbError } = await supabaseClient.from('resources').insert([ { title, description, folder_id, file_url, status: 'approved' } ]); if (dbError) { showMessage("Error al registrar el documento en la base de datos: " + dbError.message, "error"); return; } showMessage("¡Archivo subido y publicado correctamente!", "success"); $("resTitle").value = ""; $("resDesc").value = ""; $("resFile").value = ""; $("resFolderSelect").value = ""; // ⚡ Actualización inmediata de la biblioteca y panel de admin await loadFoldersAndResources(); if (currentUserRole === 'admin') await loadAdminManagementLists(); }; // 3. Al enviar una pregunta al chat, la recarga ocurre al instante window.sendQuestion = async function() { const qText = $("questionText").value.trim(); if (!qText) return; const { error } = await supabaseClient.from('questions').insert([{ question: qText }]); if (error) { showMessage("Error al enviar la pregunta", "error"); return; } $("questionText").value = ""; showMessage("¡Pregunta enviada al chat con éxito!", "success"); // ⚡ Actualización inmediata del chat await loadChatQuestions(); };
